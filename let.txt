<script>function showStatus(statusMsg, msgClass = "") {
    const statusSpan = document.getElementById("statusMessage");
    
    // 1. If the message is empty, clear the box and hide it
    if (!statusMsg || statusMsg.trim() === "") {
        statusSpan.innerHTML = "";
        statusSpan.className = "errorMsg"; 
        return;
    }
    
    // 2. Set the dynamic text safely (prevents XSS vulnerabilities)
    statusSpan.textContent = statusMsg;
    
    // 3. Apply the default class, plus any extra style class if provided
    if (msgClass !== "") {
        statusSpan.className = `errorMsg ${msgClass}`;
    } else {
        statusSpan.className = "errorMsg";
    }
}
